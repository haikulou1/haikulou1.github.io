# Login Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建一个生产级用户登录后端服务，提供登录/退出 API，支持密码加盐哈希认证、防账号枚举、暴力破解防护（失败锁定+验证码+IP限流）、安全的登录态 Cookie 管理（HttpOnly+Secure+SameSite，记住我区分长短有效期）。

**Architecture:** 采用 FastAPI 异步分层架构：路由层（API）→ 服务层（认证核心逻辑）→ 仓储层（数据访问）→ 安全层（密码哈希、限流、锁定、Cookie）。会话 token 使用 `secrets.token_urlsafe` 生成并存储于 Redis（可撤销），退出时删除 Redis key 使 token 立即失效。账号不存在与密码错误统一返回「账号或密码错误」以防止账号枚举。

**Tech Stack:** Python 3.11+, FastAPI, Uvicorn, SQLAlchemy 2.0 (async), aiosqlite (测试) / asyncpg (生产 PostgreSQL), redis-py (async) + fakeredis (测试), passlib[bcrypt], pydantic-settings, pytest, pytest-asyncio, httpx

## Global Constraints

- 密码存储使用 bcrypt 加盐哈希，`bcrypt_rounds = 12`（OWASP 推荐下限）
- 账号不存在与密码错误必须统一返回 `账号或密码错误`，禁止差异化错误信息（防账号枚举）
- 失败次数阈值 `max_failed_attempts = 5`，超过后锁定 `lockout_duration_minutes = 15` 分钟
- 失败次数达到 `captcha_threshold = 3` 时触发验证码要求
- IP 限流 `ip_rate_limit_per_minute = 10`，超限返回 HTTP 429
- 登录态 Cookie 必须 `HttpOnly=True` + `Secure=True` + `SameSite=Lax`
- 「记住我」未勾选时短期有效期 `access_token_expire_minutes = 30`，勾选时长期有效期 `refresh_token_expire_days = 30`
- 不包含注册、找回密码、SSO、第三方登录、扫码、MFA（本期排除，列入后续规划）
- 所有新增代码位于 `auth-service/` 子目录，不修改现有静态网站代码
- 严格 TDD：每个 Task 先写失败测试，再写最小实现，再验证通过
- 文件路径基于仓库根目录（cwd）写入

```python
# auth-service/src/security/password.py
from passlib.context import CryptContext

_pwd_context = CryptContext(schemes=["bcrypt"], bcrypt__rounds=12, deprecated="auto")


def hash_password(plain: str) -> str:
    return _pwd_context.hash(plain)


def verify_password(plain: str, hashed: str) -> bool:
    return _pwd_context.verify(plain, hashed)
```

```python
# auth-service/src/repositories/user_repository.py
from datetime import datetime

from sqlalchemy import select, update
from sqlalchemy.ext.asyncio import AsyncSession

from src.models.user import User


async def get_by_username(session: AsyncSession, username: str) -> User | None:
    result = await session.execute(select(User).where(User.username == username))
    return result.scalar_one_or_none()


async def increment_failed_attempts(session: AsyncSession, user_id: int) -> User:
    await session.execute(
        update(User).where(User.id == user_id).values(failed_login_count=User.failed_login_count + 1)
    )
    await session.commit()
    result = await session.execute(select(User).where(User.id == user_id))
    return result.scalar_one()


async def reset_failed_attempts(session: AsyncSession, user_id: int) -> User:
    await session.execute(
        update(User).where(User.id == user_id).values(failed_login_count=0, locked_until=None)
    )
    await session.commit()
    result = await session.execute(select(User).where(User.id == user_id))
    return result.scalar_one()


async def lock_account(session: AsyncSession, user_id: int, until: datetime) -> User:
    await session.execute(
        update(User).where(User.id == user_id).values(locked_until=until)
    )
    await session.commit()
    result = await session.execute(select(User).where(User.id == user_id))
    return result.scalar_one()


async def disable_account(session: AsyncSession, user_id: int) -> User:
    await session.execute(
        update(User).where(User.id == user_id).values(status="disabled")
    )
    await session.commit()
    result = await session.execute(select(User).where(User.id == user_id))
    return result.scalar_one()
```

```python
# auth-service/src/security/lockout.py
from dataclasses import dataclass
from redis.asyncio import Redis


@dataclass
class LockoutState:
    is_locked: bool
    remaining_seconds: int


@dataclass
class FailedAttemptResult:
    count: int
    should_lock: bool
    require_captcha: bool


async def check_lockout(redis: Redis, username: str, lockout_seconds: int) -> LockoutState:
    key = f"lockout:{username}"
    ttl = await redis.ttl(key)
    if ttl and ttl > 0:
        return LockoutState(is_locked=True, remaining_seconds=ttl)
    return LockoutState(is_locked=False, remaining_seconds=0)


async def record_failed_attempt(
    redis: Redis,
    username: str,
    max_attempts: int,
    lockout_seconds: int,
    captcha_threshold: int,
) -> FailedAttemptResult:
    count_key = f"failcount:{username}"
    count = await redis.incr(count_key)
    await redis.expire(count_key, lockout_seconds)

    require_captcha = count >= captcha_threshold
    should_lock = count >= max_attempts

    if should_lock:
        lock_key = f"lockout:{username}"
        await redis.set(lock_key, "1", ex=lockout_seconds)

    return FailedAttemptResult(
        count=count,
        should_lock=should_lock,
        require_captcha=require_captcha,
    )


async def check_captcha_required(redis: Redis, username: str, captcha_threshold: int) -> bool:
    count = await redis.get(f"failcount:{username}")
    if count is None:
        return False
    return int(count) >= captcha_threshold


async def reset_failures(redis: Redis, username: str) -> None:
    await redis.delete(f"failcount:{username}")
    await redis.delete(f"lockout:{username}")
```

```python
# auth-service/src/services/auth_service.py
import secrets
from dataclasses import dataclass
from datetime import datetime, timedelta, timezone

from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from src.config import Settings
from src.repositories.user_repository import (
    get_by_username,
    increment_failed_attempts,
    reset_failed_attempts,
    lock_account,
)
from src.security.lockout import (
    check_lockout,
    check_captcha_required,
    record_failed_attempt,
    reset_failures,
)
from src.security.password import verify_password


@dataclass
class AuthResult:
    success: bool
    message: str
    require_captcha: bool
    token: str | None
    user_id: int | None


async def authenticate(
    session: AsyncSession,
    redis: Redis,
    username: str,
    password: str,
    captcha: str | None,
    settings: Settings,
) -> AuthResult:
    lockout_seconds = settings.lockout_duration_minutes * 60

    # 1. 检查锁定状态
    lockout = await check_lockout(redis, username, lockout_seconds)
    if lockout.is_locked:
        return AuthResult(
            success=False,
            message=f"账号已锁定，请 {lockout.remaining_seconds // 60} 分钟后再试",
            require_captcha=False,
            token=None,
            user_id=None,
        )

    # 2. 检查是否需要验证码（失败次数达阈值但尚未锁定）
    captcha_required = await check_captcha_required(
        redis, username, settings.captcha_threshold
    )
    if captcha_required and not captcha:
        return AuthResult(
            success=False,
            message="请输入验证码",
            require_captcha=True,
            token=None,
            user_id=None,
        )

    # 3. 查询用户
    user = await get_by_username(session, username)

    # 4. 统一处理：账号不存在 vs 密码错误 -> 相同消息
    password_ok = user is not None and verify_password(password, user.password_hash)

    if user is not None and user.status == "disabled":
        return AuthResult(
            success=False,
            message="账号已禁用",
            require_captcha=False,
            token=None,
            user_id=None,
        )

    if not password_ok:
        attempt = await record_failed_attempt(
            redis, username,
            settings.max_failed_attempts,
            lockout_seconds,
            settings.captcha_threshold,
        )
        if user is not None:
            await increment_failed_attempts(session, user.id)
            if attempt.should_lock:
                until = datetime.now(timezone.utc) + timedelta(seconds=lockout_seconds)
                await lock_account(session, user.id, until)
        return AuthResult(
            success=False,
            message="账号或密码错误",
            require_captcha=attempt.require_captcha,
            token=None,
            user_id=None,
        )

    # 5. 密码正确 -> 成功
    await reset_failures(redis, username)
    await reset_failed_attempts(session, user.id)

    token = secrets.token_urlsafe(32)
    token_ttl = settings.access_token_expire_minutes * 60
    await redis.set(f"session:{token}", str(user.id), ex=token_ttl)

    return AuthResult(
        success=True,
        message="登录成功",
        require_captcha=False,
        token=token,
        user_id=user.id,
    )
```

> **注意：** `authenticate` 内部生成 token 并写入 Redis，但 `remember_me` 逻辑（选择 TTL）由 Task 7 路由层在调用 `authenticate` 后根据请求参数设置最终 Cookie max_age。如需 remember_me 区分 TTL，路由层在收到 token 后重新 `redis.set` 覆盖 TTL。此处保持服务层简洁。

```python
from fastapi import APIRouter, Request, Response, HTTPException, status, Depends
from redis.asyncio import Redis
from sqlalchemy.ext.asyncio import AsyncSession

from src.config import Settings, get_settings
from src.schemas.auth import LoginRequest, LoginResponse, LogoutResponse
from src.security.cookie import build_cookie_params
from src.security.rate_limiter import check_ip_rate, record_ip_request
from src.services.auth_service import authenticate

router = APIRouter(prefix="/api", tags=["auth"])


async def _get_db():
    from src.database import async_session_factory
    async with async_session_factory() as session:
        yield session


async def _get_redis():
    from redis.asyncio import Redis as _R
    redis = _R.from_url(get_settings().redis_url)
    yield redis
    await redis.aclose()


@router.post("/login", response_model=LoginResponse)
async def login(
    body: LoginRequest,
    request: Request,
    response: Response,
    settings: Settings = Depends(get_settings),
    db: AsyncSession = Depends(_get_db),
    redis: Redis = Depends(_get_redis),
):
    ip = request.client.host if request.client else "unknown"

    # IP 限流
    await record_ip_request(redis, ip)
    if not await check_ip_rate(redis, ip, settings.ip_rate_limit_per_minute):
        raise HTTPException(
            status_code=status.HTTP_429_TOO_MANY_REQUESTS,
            detail="请求过于频繁，请稍后再试",
        )

    result = await authenticate(
        db, redis, body.username, body.password, body.captcha, settings
    )

    if result.success:
        # remember_me 为 True 时覆盖 Redis TTL 为长期
        if body.remember_me:
            long_ttl = settings.refresh_token_expire_days * 86400
            await redis.set(f"session:{result.token}", str(result.user_id), ex=long_ttl)
        cookie_params = build_cookie_params(settings, body.remember_me)
        response.set_cookie(value=result.token, **cookie_params)
        return LoginResponse(
            success=True, require_captcha=False, message=result.message
        )

    # 失败时不设置 Cookie
    if result.require_captcha:
        response.status_code = status.HTTP_429_TOO_MANY_REQUESTS
    else:
        response.status_code = status.HTTP_401_UNAUTHORIZED
    return LoginResponse(
        success=False,
        require_captcha=result.require_captcha,
        message=result.message,
    )


@router.post("/logout", response_model=LogoutResponse)
async def logout(
    request: Request,
    response: Response,
    settings: Settings = Depends(get_settings),
    redis: Redis = Depends(_get_redis),
):
    token = request.cookies.get("auth_token")
    if token:
        await redis.delete(f"session:{token}")
    response.delete_cookie(
        key="auth_token",
        httponly=True,
        secure=settings.cookie_secure,
        samesite=settings.cookie_samesite,
        domain=settings.cookie_domain,
    )
    return LogoutResponse(success=True)
```

```python
from fastapi import Depends, HTTPException, Request, status
from redis.asyncio import Redis


async def _get_redis():
    from redis.asyncio import Redis as _R
    from src.config import get_settings
    redis = _R.from_url(get_settings().redis_url)
    yield redis
    await redis.aclose()


async def get_current_user(
    request: Request,
    redis: Redis = Depends(_get_redis),
) -> dict:
    token = request.cookies.get("auth_token")
    if not token:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="未登录")

    user_id_raw = await redis.get(f"session:{token}")
    if user_id_raw is None:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="登录已过期")

    return {"user_id": int(user_id_raw)}
```
